cmake_minimum_required(VERSION 3.22.1)
project(acousticsense_audio LANGUAGES CXX)
find_package(oboe REQUIRED CONFIG)
add_library(acousticsense_audio SHARED AudioMetrics.cpp TestSignal.cpp DuplexEngine.cpp jni_bridge.cpp)
target_compile_features(acousticsense_audio PRIVATE cxx_std_17)
target_compile_options(acousticsense_audio PRIVATE -Wall -Wextra -Werror)
target_link_libraries(acousticsense_audio PRIVATE oboe::oboe android log)
if(NOT ANDROID)
    enable_testing()
    add_executable(test_signal_test TestSignal.cpp TestSignalTest.cpp)
    add_test(NAME test_signal COMMAND test_signal_test)
endif()
